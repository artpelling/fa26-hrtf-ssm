\begin{frame}[t]{State-Space Realizations}
  \begin{columns}[T,onlytextwidth]
    \column{.54\textwidth}
    Discrete-time state equations:\\
    \begin{center}
      \ttcbox{\(\begin{aligned}
                  x_{k+1} & = \matA x_k + \matB u_k, \\
                  y_k     & = \matC x_k + \matD u_k.
                \end{aligned}\)}
    \end{center}
    \[
      \matA \in \R[n][n],~\matB
            \in \R[n][m],~ \matC
            \in \R[p][n],~\matD
            \in \R[p][m]
    \]
    \column{.41\textwidth}
    \centering
    \systemmatrix*{4}{3}{2}
  \end{columns}

  \vfill
  \only<2->{
    \begin{itemize}
      \item Impulse response:
            \[
              h_k = \begin{cases}
                      \matD,                 & k = 0,    \\
                      \matC\matA^{k-1}\matB, & k \geq 1.
                    \end{cases}
            \]
      \item Transfer function:
            \[
              \matH(z) = \matC(z\matI_n - \matA)^{-1}\matB + \matD
            \]
    \end{itemize}
  }
\end{frame}

\begin{frame}
  \frametitle{Hankel Matrix}
  MIMO impulse response \(h_k \in \R^{p \times m}\)
  \begin{equation*}
    \H = \begin{bmatrix}
           h_1    & h_2     & h_3     & \cdots   & h_s      \\
           h_2    & \iddots &         & \iddots  & h_{s+1}  \\
           h_3    &         & \iddots & \iddots  & \vdots   \\
           \vdots & \iddots & \iddots &          & h_{2s-2} \\
           h_s    & h_{s+1} & \cdots  & h_{2s-2} & h_{2s-1}
         \end{bmatrix}
       \in \R^{ps \times ms}
  \end{equation*}
  \vfill
  \begin{center}
    \ttcbox[fontupper=\large]{McMillan degree: \(\rank{\H}\)}
  \end{center}
\end{frame}

\begin{frame}{How to obtain a state-space realization from impulse responses?}

  \begin{itemize}
    \item Form Hankel matrix \(\H\)
    \item Compute (truncated) SVD of \(\H\)
    \item Solve a least-squares problem of size \((2s-2)r^2\)
  \end{itemize}
\end{frame}

\begin{frame}{Loss-free Virtual Input Projection}
  Tangential Eigensystem Realization Algorithm\footfullcite{kramer2016}.
  Assume \(h_k=0\) for \(k>s\).
  \[
    \Theta_R = \begin{bmatrix}h_1 \\ \vdots \\ h_s\end{bmatrix}
             = \bar{\matU}\bar{\matSigma}\bar{\matV}^\top \in \R^{ps \times m},
    \qquad q = \rank{\Theta_R}
  \]
  \[
    \matW_2 = \bar{\matV} \in \R^{m \times q},
    \quad \matW_2^\top\matW_2=\matI_q,
    \quad \hat h_k = h_k\matW_2
  \]
  \begin{center}
    \begin{tikzpicture}[x=.0009cm, y=.0009cm, baseline=(current bounding box.center)]
      \path[draw, line width=.3pt, fill=TU-hellgrau!25]
        (0,0) rectangle (11950,2);
      \path[draw, line width=.3pt, fill=TU-hellgrau!25]
        (12600,0) rectangle (13112,2);
      \node[anchor=south] at (5975,110) {\(h_k\)};
      \node[anchor=south] at (12856,110) {\(\hat h_k\)};
      \node[font=\scriptsize, anchor=north] at (5975,-110)
        {\(2 \times 11\,950\)};
      \node[font=\scriptsize, anchor=north] at (12856,-110)
        {\(2 \times 512\)};
    \end{tikzpicture}
  \end{center}
  \only<2->{
    \begin{align*}
      \matP_2 & = \matW_2\matW_2^\top,
        \quad h_k = h_k\matP_2 = \hat h_k\matW_2^\top,\quad k \geq 1, \\
      \hat{\H} & = \H(\matI_s \otimes \matW_2)
        \quad\Longrightarrow\quad \sigma_i(\hat{\H}) = \sigma_i(\H),
        \quad i=1,\ldots,\rank{\H}.
    \end{align*}
  }
\end{frame}

\begin{frame}{How to obtain a state-space realization from impulse responses?}
  \blue{Loss-free Projected} Eigensystem Realization Algorithm (for dummies)
  \begin{itemize}
    \item \blue{Calculate SVD of \(\Theta_R\) and form \(\matW_2\)}
    \item \blue{Project impulse responses with \(\matW_2\)}
    \item Form \blue{projected} Hankel matrix \blue{\(\hat\H\)}
    \item Compute (truncated) SVD of \blue{\(\hat\H\)}
    \item Solve a least-squares problem of size \((2s-2)r^2\)
    \item \blue{Backproject \(\hat\matB\matW_2^\top=\matB\)}
  \end{itemize}
  \vfill
  One SVD \(512\times3\,059\,200\) \(\quad\Rightarrow\quad\) two SVDs \(512\times11\,950\) and \(512\times131\,072\)

\begin{frame}{State Transformations}
  Let \(\matT \in \R[n][n]\) be invertible and define
  \[
    \widetilde{x}_k = \matT^{-1}x_k.
  \]
  \[
    \begin{aligned}
      \widetilde{\matA} & = \matT^{-1}\matA\matT, & \widetilde{\matB} & = \matT^{-1}\matB, \\
      \widetilde{\matC} & = \matC\matT,           & \widetilde{\matD} & = \matD.
    \end{aligned}
  \]
  \vfill
  \begin{itemize}
    \item Impulse response:
          \begin{align*}
            \widetilde{h}_k & = \widetilde{\matC}\widetilde{\matA}^{k-1}\widetilde{\matB},\quad k \geq 1 \uncover < 2 - > {\\
                                                                                                            & = \matC\blue{\matT}(\blue{\matT^{-1}}\matA\blue{\matT})^{k-1}\blue{\matT^{-1}}\matB \\
                                                                                                            & = \matC\matA^{k-1}\matB}
          \end{align*}
    \item Transfer function:
          \begin{align*}
            \widetilde{\matH}(z) & = \widetilde{\matC}(z\matI_n - \widetilde{\matA})^{-1}\widetilde{\matB} + \widetilde{\matD} \\
            \uncover < 2 - > {& = \matC\blue{\matT}(z\matI_n - \blue{\matT^{-1}}\matA\blue{\matT})^{-1}\blue{\matT^{-1}}\matB + \matD \\
                              & = \matC(z\matI_n - \matA)^{-1}\matB + \matD}
          \end{align*}
  \end{itemize}
\end{frame}
