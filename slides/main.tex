\begin{frame}{Linear Time-Invariant Systems\only<2>{ - discrete-time}}
  \only<1>
  {
    Let $u,y \in \mathcal{L}_2(\R)$, a system
    \(\Sigma:\mathcal{L}_2(\R) \rightarrow \mathcal{L}_2(\R)\)
    is
    a mapping $\Sigma(u) \mapsto y$
    \begin{figure}
      \centering \includegraphics[width=0.35\textwidth]{figures/system}
    \end{figure}
    \begin{itemize}
      \item Linearity:
            \[
              \Sigma(\alpha u_1 + u_2) = \alpha\Sigma(u_1)
                                         + \Sigma(u_2),\quad\alpha
                                       \in \R
            \]
      \item Time-invariance:
            \[
              \Sigma\mathcal{S}_{\tau}(u) = \mathcal{S}_{\tau}\Sigma(u)
            \]
    \end{itemize}
    Dynamics of LTI systems are fully described by impulse response
    $h = \Sigma(\delta) \in \mathcal{L}_2(\R)$:
    \begin{itemize}
      \item Convolution:
            \[
              \Sigma(u(t)) = (h \ast u)(t)
                           = \int_{-\infty}^\infty h(t - \tau)u(\tau)\,\mathrm{d}\tau
                           = y(t)
            \]
    \end{itemize}
  }
  \only<2>
  {
    Let $u,y \in \blue{\ell_2(\mathbb{Z})}$, a system
    \(\Sigma:\blue{\ell_2(\mathbb{Z})} \rightarrow \blue{\ell_2(\mathbb{Z})}\)
    is a mapping $\Sigma(u) \mapsto y$
    \begin{figure}
      \centering \includegraphics[width=0.35\textwidth]{figures/system}
    \end{figure}
    \begin{itemize}
      \item Linearity:
            \[
              \Sigma(\alpha u_1 + u_2) = \alpha\Sigma(u_1)
                                         + \Sigma(u_2),\quad\alpha
                                       \in \R
            \]
      \item Time-invariance:
            \[
              \Sigma\mathcal{S}_{\tau}(u) = \mathcal{S}_{\tau}\Sigma(u)
            \]
    \end{itemize}
    Dynamics of LTI systems are fully described by impulse response
    $h = \Sigma(\delta) \in \blue{\ell_2(\mathbb{Z})}$:
    \begin{itemize}
      \item Convolution:
            \[
              y_k = \Sigma(u)_k
                  = \blue{\sum_{\ell\in\N_0} h_\ell u_{k-\ell}}
            \]
    \end{itemize}
  }

\begin{frame}
  \frametitle{Frequency Domain Description} \(z\)-transform of impulse response:
  \[
    \mathcal{Z}\{h_k\} = \matH(z) = \sum_{k \in \mathbb{Z}} h_k z^{-k}
  \]
  Transmission is given by the transfer function \(\matH(z)\):
  \[
    \matY(z) = \mathcal{Z}\{y_k\}
             = \matH(z)\matU(z)
  \]
  Common representations:
  \begin{itemize}
    \item Rational:
          \[
            \matH(z)
            = \frac{\sum_{k=0}^{N}b_kz^{-k}}{1 - \sum_{k=1}^{M}a_{k}z^{-k}} \onslide
            < 2
            > {\quad \rightarrow \quad\text{barycentric form}:~\frac{\sum_{k=0}^{K}\frac{\alpha_{k}}{z - t_k}}{\sum_{k=0}^K\frac{\beta_k}{z - t_k}}}
          \]
          \onslide<2>{\footcite{trefethen2013}}
    \item Pole-residue:
          \[
            \matH(z) = \sum_{k=1}^{K}\frac{c_kb_k}{z - \lambda_k}
          \]
  \end{itemize}
\end{frame}

\begin{frame}{State-Space Realizations}
  \begin{itemize}
    \item \textbf{External description} (\(h_k, \matH(z)\))\hfill vs.\hfill \textbf{internal description} (state \(x_k\))
  \end{itemize}
  \begin{ttcbox}
    Discrete-time state equations:
    \[
      \begin{gathered}
        x_{k+1} = \matA x_k + \matB u_k, \qquad
        y_k = \matC x_k + \matD u_k, \\
        \matA \in \R[n][n],\quad \matB \in \R[n][m], \\
        \matC \in \R[p][n],\quad \matD \in \R[p][m]
      \end{gathered}
    \]
  \end{ttcbox}
  \begin{itemize}
    \item Markov parameters: \(h_0 = \matD\), \(h_k = \matC\matA^{k-1}\matB\) for \(k \geq 1\)
    \item Transfer function: \(\matH(z) = \matC(z\matI_n - \matA)^{-1}\matB + \matD\)
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{McMillan degree bound}
  \begin{theorem}
    \label{thm:order-bound}
    Let \(h_k \in \R[p][m]\), \(k = 0,\dots,s\), denote an FIR-approximated
    impulse response sequence with \(h_k = 0\) for every \(k > s\). Then, the
    rational transfer function given by \(\matH_s(z) = \sum_{k=0}^sh_kz^{-k}\)
    has McMillan degree at most \(ps\). In particular, for a MIMO-HRTF,
    \(p = 2\) (the two ears), so its McMillan degree is at most \(2s\),
    independently of the number \(m\) of sampled directions.
  \end{theorem}
\end{frame}
%
\begin{frame}
  \frametitle{Loss-free virtual inputs}
  \begin{theorem}
    \label{thm:virtual inputs}
    Assume \(h_k = 0\) for every \(k > s\). Then, \(q \leq \rank{\H} \leq ps\)
    and \(h_k = h_k\matP_2\) for every \(k \geq 1\). Consequently, the projected
    Hankel matrix constructed from \(\hat h_k = h_k\matW_2\) has the same
    nonzero Hankel singular values as \(\H\). For every reduced order, the
    Markov parameters of the projected and unprojected ERA models are identical.
  \end{theorem}
\end{frame}
